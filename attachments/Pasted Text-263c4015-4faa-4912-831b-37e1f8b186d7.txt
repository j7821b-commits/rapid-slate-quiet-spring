        function addLogEntry(message, type = 'info') {
            const logContainer = document.getElementById('swarm-event-log');
            const entry = document.createElement('div');
            
            let icon = 'fa-info-circle text-teal-400';
            let color = 'text-slate-300';
            
            if (type === 'action') {
                icon = 'fa-bolt text-yellow-400';
                color = 'text-white font-semibold';
            } else if (type === 'human') {
                icon = 'fa-user-check text-violet-400';
                color = 'text-violet-200';
            } else if (type === 'success') {
                icon = 'fa-check-circle text-emerald-400';
                color = 'text-emerald-300';
            }

            const time = new Date().toLocaleTimeString([], {hour: '2-digit', minute:'2-digit', second:'2-digit'});

            entry.className = `text-sm mb-3 border-l-2 border-slate-700 pl-2 py-1`;
            entry.innerHTML = `
                <div class="flex items-start">
                    <i class="fa-solid ${icon} mt-1 mr-2 w-4"></i>
                    <div>
                        <span class="text-xs text-slate-500 mr-2">[${time}]</span>
                        <span class="${color}">${message}</span>
                    </div>
                </div>
            `;
            
            logContainer.appendChild(entry);
            logContainer.scrollTop = logContainer.scrollHeight;
        }

        function triggerSwarmAction(unit) {
            if (state[unit].resolved) return;
            
            const btn = document.getElementById(`btn-swarm-${unit.toLowerCase()}`);
            btn.disabled = true;
            btn.innerHTML = '<i class="fa-solid fa-spinner fa-spin mr-2"></i>Swarm Processing...';
            btn.classList.replace('bg-teal-600', 'bg-slate-600');

                    document.getElementById('analog-status-a').classList.remove('hidden');
                    btn.style.display = 'none';
                }, 4000);

            } else if (unit === 'B') {
                // Unit B Simulation: Utility Assistance (Analog Ministry - CAM)

                setTimeout(() => {

                    document.getElementById('analog-status-b').classList.remove('hidden');
                    btn.style.display = 'none';
                }, 4000);
            }
        }
